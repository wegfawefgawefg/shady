cmake_minimum_required(VERSION 3.20)
project(shady VERSION 0.1.0 LANGUAGES CXX)

option(SHADY_STRICT "Enable strict compiler warnings" ON)
option(SHADY_WARN_AS_ERROR "Treat warnings as errors" OFF)
option(SHADY_BUILD_DESKTOP "Build the SDL desktop runner" ON)
option(SHADY_BUILD_FRAME "Build the headless frame renderer" ON)
option(SHADY_BUILD_TESTS "Build tests" ON)
set(SHADY_WEBGPU_DISTRIBUTION_TAG "main-v0.2.0" CACHE STRING "WebGPU-distribution tag")
set(SHADY_WEBGPU_BACKEND "WGPU" CACHE STRING "Native WebGPU backend")
set_property(CACHE SHADY_WEBGPU_BACKEND PROPERTY STRINGS WGPU WGPU_STATIC DAWN)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

function(shady_warnings target)
    if(NOT SHADY_STRICT)
        return()
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
        if(SHADY_WARN_AS_ERROR)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(SHADY_WARN_AS_ERROR)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()

find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SDL2_IMAGE REQUIRED IMPORTED_TARGET SDL2_image)

include(FetchContent)
set(WEBGPU_BACKEND "${SHADY_WEBGPU_BACKEND}" CACHE STRING "WebGPU backend" FORCE)
set(WEBGPU_BUILD_FROM_SOURCE OFF CACHE BOOL "Use packaged WebGPU" FORCE)
set(WEBGPU_LINK_TYPE SHARED CACHE STRING "WebGPU link type" FORCE)
FetchContent_Declare(shady_webgpu
    URL "https://github.com/eliemichel/WebGPU-distribution/archive/refs/tags/${SHADY_WEBGPU_DISTRIBUTION_TAG}.zip"
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(shady_webgpu)

add_library(shady_support STATIC src/arguments.cpp src/channels.cpp src/files.cpp src/media.cpp)
add_library(shady::support ALIAS shady_support)
target_include_directories(shady_support PUBLIC include)
target_compile_definitions(shady_support PRIVATE SHADY_RUNTIME_DIR="${CMAKE_CURRENT_SOURCE_DIR}/runtime")
target_link_libraries(shady_support PUBLIC PkgConfig::SDL2 PkgConfig::SDL2_IMAGE)
shady_warnings(shady_support)

add_library(shady_gpu STATIC src/gpu.cpp)
add_library(shady::gpu ALIAS shady_gpu)
target_include_directories(shady_gpu PUBLIC include)
target_link_libraries(shady_gpu PUBLIC shady::support webgpu)
shady_warnings(shady_gpu)

if(SHADY_BUILD_FRAME)
    add_executable(shady-frame src/frame.cpp)
    target_link_libraries(shady-frame PRIVATE shady::gpu)
    shady_warnings(shady-frame)
    if(COMMAND target_copy_webgpu_binaries)
        target_copy_webgpu_binaries(shady-frame)
    endif()
endif()

if(SHADY_BUILD_DESKTOP AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/app.cpp")
    add_executable(shady src/app.cpp src/presentation.cpp src/live_media.cpp)
    target_link_libraries(shady PRIVATE shady::gpu)
    shady_warnings(shady)
    if(COMMAND target_copy_webgpu_binaries)
        target_copy_webgpu_binaries(shady)
    endif()
endif()

if(SHADY_BUILD_TESTS)
    enable_testing()
    add_executable(shady_tests tests/shady_tests.cpp)
    target_link_libraries(shady_tests PRIVATE shady::support)
    shady_warnings(shady_tests)
    add_test(NAME shady_tests COMMAND shady_tests)
endif()
